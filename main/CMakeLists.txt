# CDC Badge OS v0.5 - Main component

# =============================================================================
# MODULE CONFIGURATION
# Add or remove module names here to enable/disable them.
# Each module must provide a <modulename>_register() function.
# =============================================================================
set(MODULES
    mod_totp
    mod_fido2
    mod_password
    mod_gpg
    mod_sao
    mod_vcard
    mod_ble_serial
    mod_nvsedit
    mod_hid
    mod_vfat
)
# =============================================================================

idf_component_register(
    SRCS
        "main.cpp"
    INCLUDE_DIRS "." "${CMAKE_CURRENT_BINARY_DIR}"
    REQUIRES nvs_flash freertos esp_timer cdc_core usb_badge serial_cmd cdc_hal cdc_log cdc_os_ui
        wamr_runtime plugin_manager
        ${MODULES}
)

# Generate modules_init.gen.h with all module registration calls
set(MODULES_HEADER "${CMAKE_CURRENT_BINARY_DIR}/modules_init.gen.h")
file(WRITE ${MODULES_HEADER} "// Auto-generated - DO NOT EDIT\n")
file(APPEND ${MODULES_HEADER} "// Module registration header generated by CMake\n")
file(APPEND ${MODULES_HEADER} "#pragma once\n\n")

# Generate extern declarations
foreach(MODULE ${MODULES})
    file(APPEND ${MODULES_HEADER} "extern \"C\" void ${MODULE}_register();\n")
endforeach()

# Generate registration function
file(APPEND ${MODULES_HEADER} "\ninline void modules_register_all() {\n")
foreach(MODULE ${MODULES})
    file(APPEND ${MODULES_HEADER} "    ${MODULE}_register();\n")
endforeach()
file(APPEND ${MODULES_HEADER} "}\n")
